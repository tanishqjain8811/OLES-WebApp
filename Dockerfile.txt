# syntax=docker/dockerfile:1

# a) Java/Maven base image for the build stage
FROM maven:3.9-eclipse-temurin-17 AS build
WORKDIR /app

# b) Copy Maven project files (pom first for dependency caching)
COPY pom.xml .
RUN mvn dependency:go-offline -B
COPY src ./src

# c) Build the Maven application and generate the WAR file
RUN mvn clean package -DskipTests

# d) Tomcat image; copy WAR into the webapps directory
FROM tomcat:10.1-jdk17-temurin
RUN rm -rf /usr/local/tomcat/webapps/*
COPY --from=build /app/target/*.war /usr/local/tomcat/webapps/ROOT.war

# e) Expose the port and start Tomcat
EXPOSE 8080
CMD ["catalina.sh", "run"]
